cmake_minimum_required(VERSION 3.20)
project(125A_Rifferator VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(rifferator_core_tests tests/core_contract_tests.cpp)
target_include_directories(rifferator_core_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
  target_compile_options(rifferator_core_tests PRIVATE /W4 /WX)
else()
  target_compile_options(rifferator_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME rifferator_core_contract COMMAND rifferator_core_tests)
# VST3 target deliberately deferred until static design and host adapter exist.

add_executable(rifferator_runtime_tests tests/midi_runtime_tests.cpp)
target_include_directories(rifferator_runtime_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
 target_compile_options(rifferator_runtime_tests PRIVATE /W4 /WX)
else()
 target_compile_options(rifferator_runtime_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME rifferator_midi_runtime COMMAND rifferator_runtime_tests)

add_executable(rifferator_midi_file_tests tests/midi_file_roundtrip_tests.cpp)
target_include_directories(rifferator_midi_file_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
 target_compile_options(rifferator_midi_file_tests PRIVATE /W4 /WX)
else()
 target_compile_options(rifferator_midi_file_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME rifferator_midi_file_format0 COMMAND rifferator_midi_file_tests)

add_executable(rifferator_state_tests tests/state_roundtrip_tests.cpp)
target_include_directories(rifferator_state_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
 target_compile_options(rifferator_state_tests PRIVATE /W4 /WX)
else()
 target_compile_options(rifferator_state_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME rifferator_state_roundtrip COMMAND rifferator_state_tests)

option(RIFFERATOR_ENABLE_VST3 "Build experimental VST3 host adapter" OFF)
if(RIFFERATOR_ENABLE_VST3)
 set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
 include(FetchContent)
 FetchContent_Declare(vst3sdk
  GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
  GIT_TAG v3.8.1_build_84
  GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(vst3sdk)
 smtg_add_vst3plugin(Rifferator src/vst3/Plugin.cpp src/vst3/Plugin.hpp src/vst3/Parameters.hpp)
 target_link_libraries(Rifferator PRIVATE sdk)
 target_include_directories(Rifferator PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
 if(WIN32)
  target_sources(Rifferator PRIVATE "${vst3sdk_SOURCE_DIR}/public.sdk/source/main/dllmain.cpp")
 endif()
 smtg_target_configure_version_file(Rifferator)
endif()

# Ensure assertions remain active in Release; never silently disable entire regression suite.
if(MSVC)
  target_compile_options(rifferator_core_tests PRIVATE /UNDEBUG)
  target_compile_options(rifferator_runtime_tests PRIVATE /UNDEBUG)
  target_compile_options(rifferator_midi_file_tests PRIVATE /UNDEBUG)
  target_compile_options(rifferator_state_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(rifferator_core_tests PRIVATE -UNDEBUG)
  target_compile_options(rifferator_runtime_tests PRIVATE -UNDEBUG)
  target_compile_options(rifferator_midi_file_tests PRIVATE -UNDEBUG)
  target_compile_options(rifferator_state_tests PRIVATE -UNDEBUG)
endif()
